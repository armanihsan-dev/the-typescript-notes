String Manipulation Utility Types in TypeScript

🎯 What Are String Manipulation Utility Types?

Definition:

String manipulation utility types are built-in TypeScript types that transform string literal types by changing their case. They work at the type level — no runtime code, just types.

The Four Utility Types

Utility Type What It Does Example
Uppercase<T> Converts to UPPERCASE Uppercase<"hello"> → "HELLO"
Lowercase<T> Converts to lowercase Lowercase<"HELLO"> → "hello"
Capitalize<T> Capitalizes first letter Capitalize<"hello"> → "Hello"
Uncapitalize<T> Uncapitalizes first letter Uncapitalize<"Hello"> → "hello"

1. 🔠 Uppercase<T> — Convert to UPPERCASE

Definition

Uppercase<T> converts a string literal type to ALL UPPERCASE.

Syntax

type Result = Uppercase<"hello">;
// "HELLO"

Examples

type A = Uppercase<"hello">;      // "HELLO"
type B = Uppercase<"Hello World">; // "HELLO WORLD"
type C = Uppercase<"abc123">;     // "ABC123"
type D = Uppercase<"">;           // ""

// ❌ Doesn't work with general string
// type E = Uppercase<string>;    // string (not transformed)

Use Case: Constant Names

type ConstantName<T extends string> = Uppercase<T>;

type APIKey = ConstantName<"apiKey">;      // "APIKEY"
type DatabaseURL = ConstantName<"databaseUrl">; // "DATABASEURL"

// Use in objects
const config = {
    APIKEY: "secret",
    DATABASEURL: "postgres://..."
} as const;

2. 🔡 Lowercase<T> — Convert to lowercase

Definition

Lowercase<T> converts a string literal type to all lowercase.

Syntax

type Result = Lowercase<"HELLO">;
// "hello"

Examples

type A = Lowercase<"HELLO">;      // "hello"
type B = Lowercase<"Hello World">; // "hello world"
type C = Lowercase<"ABC123">;     // "abc123"
type D = Lowercase<"">;           // ""

Use Case: CSS Class Names

type CssClass<T extends string> = Lowercase<T>;

type ButtonClass = CssClass<"PRIMARY">;    // "primary"
type DangerClass = CssClass<"DANGER">;     // "danger"

// Use in components
const button = document.createElement("button");
button.className = "primary"; // ✅

3. 🔤 Capitalize<T> — Capitalize First Letter

Definition

Capitalize<T> converts the first letter of a string literal type to uppercase.

Syntax

type Result = Capitalize<"hello">;
// "Hello"

Examples

type A = Capitalize<"hello">;      // "Hello"
type B = Capitalize<"hello world">; // "Hello world"
type C = Capitalize<"helloWorld">; // "HelloWorld"
type D = Capitalize<"">;           // ""

Use Case: Event Handler Names

type EventHandlerName<T extends string> = `on${Capitalize<T>}`;

type ClickHandler = EventHandlerName<"click">;   // "onClick"
type HoverHandler = EventHandlerName<"hover">;   // "onHover"
type FocusHandler = EventHandlerName<"focus">;   // "onFocus"

// Use in React components
interface ButtonProps {
    onClick?: () => void;
    onHover?: () => void;
    onFocus?: () => void;
}

Use Case: Getter Names

type GetterName<T extends string> = `get${Capitalize<T>}`;

type NameGetter = GetterName<"name">;   // "getName"
type AgeGetter = GetterName<"age">;     // "getAge"
type EmailGetter = GetterName<"email">; // "getEmail"

4. 🔤 Uncapitalize<T> — Uncapitalize First Letter

Definition

Uncapitalize<T> converts the first letter of a string literal type to lowercase.

Syntax

type Result = Uncapitalize<"Hello">;
// "hello"

Examples

type A = Uncapitalize<"Hello">;      // "hello"
type B = Uncapitalize<"Hello World">; // "hello World"
type C = Uncapitalize<"HelloWorld">; // "helloWorld"
type D = Uncapitalize<"">;           // ""

Use Case: Setter Names

type SetterName<T extends string> = `set${Capitalize<Uncapitalize<T>>}`;

type NameSetter = SetterName<"Name">;   // "setName"
type AgeSetter = SetterName<"Age">;     // "setAge"
type EmailSetter = SetterName<"Email">; // "setEmail"

🧠 Combining String Manipulation Types

Example 1: Event Handler System

type EventName = "click" | "hover" | "focus";

type EventHandler<T extends string> = {
    [K in T as `on${Capitalize<K>}`]: () => void;
};

type Handlers = EventHandler<EventName>;
// {
//     onClick: () => void;
//     onHover: () => void;
//     onFocus: () => void;
// }

const handlers: Handlers = {
    onClick: () => console.log("Clicked"),
    onHover: () => console.log("Hovered"),
    onFocus: () => console.log("Focused")
};

Example 2: Type-Safe Getters

type Getters<T> = {
    [K in keyof T as `get${Capitalize<string & K>}`]: () => T[K];
};

interface User {
    name: string;
    age: number;
    email: string;
}

type UserGetters = Getters<User>;
// {
//     getName: () => string;
//     getAge: () => number;
//     getEmail: () => string;
// }

Example 3: Type-Safe Setters

type Setters<T> = {
    [K in keyof T as `set${Capitalize<string & K>}`]: (value: T[K]) => void;
};

type UserSetters = Setters<User>;
// {
//     setName: (value: string) => void;
//     setAge: (value: number) => void;
//     setEmail: (value: string) => void;
// }

Example 4: CSS Class Names

type Size = "sm" | "md" | "lg";
type Color = "primary" | "secondary" | "danger";

type ButtonClass = `btn-${Lowercase<Color>}-${Lowercase<Size>}`;
// "btn-primary-sm" | "btn-primary-md" | "btn-primary-lg" |
// "btn-secondary-sm" | ...

function getButtonClass(color: Color, size: Size): ButtonClass {
    return `btn-${color.toLowerCase()}-${size.toLowerCase()}` as ButtonClass;
}

🧠 Real-World Example: API Route Builder

type HttpMethod = "get" | "post" | "put" | "delete";

type RouteBuilder<Method extends HttpMethod, Path extends string> = {
    [K in Method as `${Uppercase<K>}_${Uppercase<Path>}`]: () => void;
};

type GetUsers = RouteBuilder<"get", "users">;
// { GET_USERS: () => void }

type PostUsers = RouteBuilder<"post", "users">;
// { POST_USERS: () => void }

🧠 Real-World Example: Type-Safe Event Emitter

type EventName = "user:login" | "user:logout" | "post:create";

type EventHandlerMap = {
    [K in EventName as `on${Capitalize<K extends `${string}:${infer E}` ? E : never>}`]: 
        (payload: any) => void;
};

type Handlers = EventHandlerMap;
// {
//     onLogin: (payload: any) => void;
//     onLogout: (payload: any) => void;
//     onCreate: (payload: any) => void;
// }
