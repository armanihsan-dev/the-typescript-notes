Part 1: Zod — Runtime Validation + Type Inference

Setup

npm install zod

Practice 1: Basic Schema & Type Inference

import { z } from "zod";

// Define the schema
const UserSchema = z.object({
    id: z.number(),
    name: z.string(),
    email: z.string().email(),
    age: z.number().min(0).max(120)
});

// Extract the TypeScript type
type User = z.infer<typeof UserSchema>;

// Now you have BOTH:
// 1. Runtime validation (UserSchema.parse)
// 2. Compile-time type (User)

const validData = {
    id: 1,
    name: "Alice",
    email: "alice@email.com",
    age: 25
};

const user: User = UserSchema.parse(validData); // ✅ Works
console.log(user.name); // "Alice"

// ❌ Runtime error (caught by Zod)
try {
    UserSchema.parse({
        id: "not a number", // Wrong type
        name: "Bob",
        email: "invalid-email", // Invalid format
        age: 200 // Too old
    });
} catch (error) {
    console.log("Validation failed!");
}

Practice 2: Nested Objects

const AddressSchema = z.object({
    street: z.string(),
    city: z.string(),
    zip: z.string().regex(/^\d{5}$/)
});

const PersonSchema = z.object({
    name: z.string(),
    address: AddressSchema, // ← Nested schema
    hobbies: z.array(z.string())
});

type Person = z.infer<typeof PersonSchema>;
// {
//     name: string;
//     address: { street: string; city: string; zip: string };
//     hobbies: string[];
// }

const person: Person = PersonSchema.parse({
    name: "Alice",
    address: {
        street: "123 Main St",
        city: "NYC",
        zip: "10001"
    },
    hobbies: ["reading", "coding"]
});

Practice 3: API Response Validation

const ApiResponseSchema = z.discriminatedUnion("status", [
    z.object({
        status: z.literal("success"),
        data: z.object({ id: z.number(), name: z.string() })
    }),
    z.object({
        status: z.literal("error"),
        message: z.string()
    })
]);

type ApiResponse = z.infer<typeof ApiResponseSchema>;

async function fetchUser(id: number): Promise<ApiResponse> {
    const response = await fetch(`/api/users/${id}`);
    const data = await response.json();
    
    // ✅ Validates at runtime AND narrows the type
    return ApiResponseSchema.parse(data);
}

// Usage
const result = await fetchUser(1);
if (result.status === "success") {
    console.log(result.data.name); // ✅ TypeScript knows this exists
} else {
    console.log(result.message); // ✅ TypeScript knows this exists
}

Practice 4: Transformations

const DateSchema = z.string().transform((str) => new Date(str));

type DateType = z.infer<typeof DateSchema>; // Date

const date = DateSchema.parse("2024-01-01");
console.log(date instanceof Date); // true

// More complex transformation
const NumberFromStringSchema = z.string().transform((str) => {
    const num = parseInt(str, 10);
    if (isNaN(num)) throw new Error("Not a number");
    return num;
});

const num = NumberFromStringSchema.parse("42"); // 42 (as number)

Practice 5: Form Validation

const FormSchema = z.object({
    email: z.string().email("Invalid email"),
    password: z.string().min(8, "Password must be at least 8 characters"),
    confirmPassword: z.string(),
    age: z.number().min(18, "Must be 18 or older")
}).refine((data) => data.password === data.confirmPassword, {
    message: "Passwords don't match",
    path: ["confirmPassword"]
});

type FormData = z.infer<typeof FormSchema>;

function validateForm(data: FormData) {
    const result = FormSchema.safeParse(data);
    if (!result.success) {
        console.log(result.error.issues); // Detailed errors
        return;
    }
    // result.data is validated
}

Practice 4: SetOptional and SetRequired

import type { SetOptional, SetRequired } from "type-fest";

type User = {
    id: number;
    name: string;
    email: string;
    age: number;
};

// Make 'age' optional
type UserWithOptionalAge = SetOptional<User, "age">;
// { id: number; name: string; email: string; age?: number }

// Make 'id' required (if it was optional)
type PartialUser = Partial<User>;
type UserWithRequiredId = SetRequired<PartialUser, "id">;
// { id: number; name?: string; email?: string; age?: number }

Practice 2: Opaque — Branded Types

import type { Opaque } from "ts-essentials";

type UserId = Opaque<"UserId", string>;
type ProductId = Opaque<"ProductId", string>;

function createUserId(id: string): UserId {
    return id as UserId;
}

function createProductId(id: string): ProductId {
    return id as ProductId;
}

const userId = createUserId("user-123");
const productId = createProductId("prod-456");

// ✅ Works
function getUser(id: UserId) {}
getUser(userId);

// ❌ Error: ProductId is not UserId
// getUser(productId);

Part 4: ts-toolbelt — Advanced Type Programming

Setup

npm install ts-toolbelt

Practice 1: Object.Merge — Deep Merge

import { Object } from "ts-toolbelt";

type A = { a: string; nested: { x: number } };
type B = { b: number; nested: { y: string } };

type Merged = Object.Merge<A, B>;
// { a: string; b: number; nested: { x: number; y: string } }

Practice 2: Union.Intersect — Intersect Unions

import { Union } from "ts-toolbelt";

type A = "a" | "b" | "c";
type B = "b" | "c" | "d";

type Intersection = Union.Intersect<A, B>;
// "b" | "c"

Practice 3: List.Head and List.Tail

import { List } from "ts-toolbelt";

type Tuple = [string, number, boolean];

type Head = List.Head<Tuple>; // string
type Tail = List.Tail<Tuple>; // [number, boolean]

Practice 4: String.Split — Split Strings at Type Level

import { String } from "ts-toolbelt";

type Path = "users/123/posts/456";
type Parts = String.Split<Path, "/">;
// ["users", "123", "posts", "456"]

Part 5: Putting It All Together — Real-World Project

Scenario: Type-Safe API Client

import { z } from "zod";
import type { Merge, PartialDeep } from "type-fest";

// ─────────────────────────────────────────────
// 1. Define schemas (runtime + type)
// ─────────────────────────────────────────────
const UserSchema = z.object({
    id: z.number(),
    name: z.string(),
    email: z.string().email(),
    profile: z.object({
        bio: z.string(),
        avatar: z.string().url()
    })
});

type User = z.infer<typeof UserSchema>;

// ─────────────────────────────────────────────
// 2. Create update payload type (deep partial)
// ─────────────────────────────────────────────
type UserUpdate = PartialDeep<Omit<User, "id">>;

// ─────────────────────────────────────────────
// 3. API client with validation
// ─────────────────────────────────────────────
async function fetchUser(id: number): Promise<User> {
    const response = await fetch(`/api/users/${id}`);
    const data = await response.json();
    return UserSchema.parse(data); // ✅ Validates
}

async function updateUser(id: number, updates: UserUpdate): Promise<User> {
    const response = await fetch(`/api/users/${id}`, {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(updates)
    });
    const data = await response.json();
    return UserSchema.parse(data); // ✅ Validates
}

// ─────────────────────────────────────────────
// 4. Usage
// ─────────────────────────────────────────────
const user = await fetchUser(1);
console.log(user.profile.bio); // ✅ Type-safe

// Update only the bio
await updateUser(1, {
    profile: {
        bio: "New bio" // ✅ PartialDeep allows nested partial
    }
});

📝 Summary Table

Library Best For Key Types

Type-Fest Deep utilities Merge, PartialDeep, ReadonlyDeep, Paths
ts-essentials Practical utilities DeepPartial, DeepReadonly, Opaque
ts-toolbelt Advanced type programming Object.Merge, Union.Intersect, List.Head

🎯 Key Takeaways

Concept Explanation

ts-essentials Branded types, deep utilities
ts-toolbelt Advanced type-level programming
Combining Use Zod for validation + Type-Fest for type manipulation
