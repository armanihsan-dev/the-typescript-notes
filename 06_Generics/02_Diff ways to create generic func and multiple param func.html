Part 1: Different Ways to Write Generic Functions

1. Function Declaration (Standard)

function identity(value: T): T {
    return value;
}

// Usage:
const result = identity("hello");
// Or with inference:
const result2 = identity("hello");

2. Function Expression

const identity = function(value: T): T {
    return value;
};

// Usage:
const result = identity("hello");

3. Arrow Function (Most Common)

const identity = (value: T): T => {
    return value;
};

// One-liner:
const identity = (value: T): T => value;

// Usage:
const result = identity("hello");

4. Arrow Function with Type Parameter Before the Arrow

// ✅ Works: T is declared before the arrow
const identity = (value: T): T => value;

// ❌ Fails: T can't be declared after the arrow
const identity = (value: T): T => value; // ❌ Error: T is not defined

5. Function in Object

const utils = {
    identity(value: T): T {
        return value;
    }
};

const result = utils.identity("hello");

6. Method in Class

class Utils {
    identity(value: T): T {
        return value;
    }
}

const utils = new Utils();
const result = utils.identity("hello");

Part 2: Multiple Generic Parameters

2.1 Basic Multiple Parameters

function pair(first: T, second: U): [T, U] {
    return [first, second];
}

// Usage with explicit types:
const result1 = pair("hello", 42);
// result1 type: [string, number]

// Usage with inference:
const result2 = pair("hello", 42);
// TypeScript infers: T = string, U = number

2.3 Multiple Parameters in Arrow Function

const pair = (first: T, second: U): [T, U] => [first, second];

// Usage:
const result = pair("hello", 42);
