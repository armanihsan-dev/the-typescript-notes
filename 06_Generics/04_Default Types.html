Part 1: What Are Default Generic Parameters?

Definition:

A default generic parameter provides a fallback type when the user doesn't specify one. If no type is provided, TypeScript uses the default.

The Core Idea

// ❌ Without default
function identity(value: T): T {
    return value;
}

identity("hello"); // ✅ T = string (inferred)

// But what if TypeScript can't infer?
function createArray(): T[] {
    return [];
}

const arr = createArray(); // ❌ Error: Could not infer T

// ✅ With default
function createArray(): T[] {
    return [];
}

const arr = createArray(); // ✅ T = string (default)
const arr2 = createArray(); // ✅ T = number (explicit)

Part 2: Basic Syntax

Syntax

function functionName(param: T): T {
    return param;
}

Simple Examples

// 1. Basic default type
function identity(value: T): T {
    return value;
}

// Usage:
const result1 = identity("hello"); // T = string (default)
const result2 = identity(42); // T = number (explicit)

Part 3: Why Use Defaults?

The Problem (Without Defaults)

// ❌ Function with no default
function createArray(): T[] {
    return [];
}

const arr1 = createArray(); // ✅ Must specify T
const arr2 = createArray(); // ❌ Error: Could not infer T

The Solution (With Defaults)

// ✅ Function with default
function createArray(): T[] {
    return [];
}

const arr1 = createArray(); // ✅ T = string (default)
const arr2 = createArray(); // ✅ T = number (explicit)

When Defaults Are Useful

Scenario Without Default With Default
No arguments❌ Must specify✅ Uses default
Common use case❌ Always explicit✅ Just works
Complex types❌ Hard to use✅ Easy to use
Libraries❌ Hard to use✅ Easy to use

Part 4: Multiple Default Parameters

Syntax

function functionName(param1: T, param2: U): [T, U] {
    return [param1, param2];
}

Examples

// 1. Two default types
function pair(first: T, second: U): [T, U] {
    return [first, second];
}

// Usage:
const p1 = pair("hello", 42); // T = string, U = number (both default)
const p2 = pair(true, 42); // T = boolean, U = number (U default)
const p3 = pair(true, "hello"); // Both explicit

// 2. Mixed defaults and inference
function createMap(): Map {
    return new Map();
}

const map1 = createMap(); // Map (both defaults)
const map2 = createMap(); // Map (V default)
const map3 = createMap(); // Map (both explicit)

Order of Precedence

┌─────────────────────────────────────────────────────────────────┐
│                    ORDER OF PRECEDENCE                          │
├─────────────────────────────────────────────────────────────────┤
│                                                                  │
│  function pair(...): [T, U]            │
│                                                                  │
│  1. Explicit type > Default type                               │
│     pair(true, 42) → T = boolean, U = number         │
│                                                                  │
│  2. Inference > Default type                                    │
│     pair("hello", 42) → T = string, U = number                │
│                                                                  │
│  3. Default used when nothing else works                       │
│     createMap() → T = string, U = number                      │
│                                                                  │
│  TL;DR: Explicit > Inference > Default                         │
│                                                                  │
└─────────────────────────────────────────────────────────────────┘

Example 1: API Response

// API response with default type
interface ApiResponse {
    status: number;
    data: T;
    message: string;
}

// Usage:
const response1: ApiResponse = {
    status: 200,
    data: "Hello", // T = any (default)
    message: "Success"
};

const response2: ApiResponse = {
    status: 200,
    data: { id: 1, name: "Alice" }, // T = User
    message: "Success"
};

// Generic function with default
function createResponse(data: T): ApiResponse {
    return {
        status: 200,
        data,
        message: "Success"
    };
}

// Usage:
const r1 = createResponse("hello"); // T = string (default)
const r2 = createResponse(42); // T = number (inferred)
const r3 = createResponse({ id: 1, name: "Alice" }); // T = User
