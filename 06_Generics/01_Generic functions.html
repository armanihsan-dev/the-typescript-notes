Part 1: What is a Generic Function?

Definition:

A generic function is a function that works with any type, while still preserving the type information of its inputs and outputs.

The Problem Without Generics:

// ❌ 'any' loses type safety
function identity(value: any): any {
    return value;
}

const result = identity("hello");
// TypeScript thinks: result is 'any'
// We lost the type information!

The Solution With Generics:

// ✅ Generic preserves type safety
function identity(value: T): T {
    return value;
}

const result = identity("hello");
// TypeScript knows: result is 'string'
// Type information is preserved!

Part 2: Syntax of Generic Functions

Basic Syntax

function functionName(parameter: T): T {
    // Function body
    return parameter;
}

Syntax Breakdown

function identity(value: T): T {
    //        ↑       ↑       ↑
    //        |       |       |
    //   Type      Param   Return
    //   Parameter  Type    Type
    return value;
}

Part 3: How Generic Functions Work

At Compile Time (TypeScript)

function identity(value: T): T {
    return value;
}

// Step 1: You call the function with a string
const result = identity("hello");

// Step 2: TypeScript infers T = string
// The type becomes: function identity(value: string): string

// Step 3: TypeScript checks: Is the argument a string? Yes.
// Step 4: TypeScript knows: result is string.

// All of this happens BEFORE the code runs!

At Runtime (JavaScript)

// After compilation, the generated JavaScript is:

function identity(value) {
    return value;
}

// ❌ The generic syntax is GONE!
// ❌ T is GONE!
// ❌ Type information is GONE!
The key insight:

Generics exist ONLY at compile time. At runtime, they don't exist at all!

Part 4: Multiple Type Parameters

Syntax

function functionName(param1: T, param2: U): [T, U] {
    return [param1, param2];
}

Example: Pair Function

function pair(first: T, second: U): [T, U] {
    return [first, second];
}

// Usage:
const result = pair("hello", 42);
// result type: [string, number]

// TypeScript can also infer both:
const result2 = pair("hello", 42);
// TypeScript knows: T = string, U = number

Example: Key-Value Function

function createEntry(key: K, value: V): { key: K; value: V } {
    return { key, value };
}

const entry = createEntry("name", "Alice");
// entry type: { key: string; value: string }

const entry2 = createEntry(1, "Bob");
// entry2 type: { key: number; value: string }

Part 5: Generic Functions with Types and Interfaces

Using Generic Types in Functions

// Define a generic type
type Box = {
    value: T;
};

// Function that uses the generic type
function createBox(value: T): Box {
    return { value };
}

const box = createBox("hello");
// box type: Box

Using Generic Interfaces in Functions

// Define a generic interface
interface Repository {
    items: T[];
    add(item: T): void;
}

// Function that creates a repository
function createRepository(): Repository {
    return {
        items: [],
        add(item: T) {
            this.items.push(item);
        }
    };
}

const stringRepo = createRepository();
stringRepo.add("hello");
stringRepo.add("world");
// stringRepo.items: string[]

const numberRepo = createRepository();
numberRepo.add(42);
// numberRepo.items: number[]

Part 6: Type Inference in Generic Functions

Automatic Inference

function identity(value: T): T {
    return value;
}

// TypeScript infers T from the argument:
const result = identity("hello"); // T = string
const result2 = identity(42);     // T = number
const result3 = identity(true);   // T = boolean

When Inference Fails

// ❌ Can't infer T (no argument)
function createEmptyArray(): T[] {
    return [];
}

const array = createEmptyArray(); // Error: Could not infer T

// ✅ Fix: Explicitly specify
const array2 = createEmptyArray(); // Works

Partial Inference

function pair(first: T, second: U): [T, U] {
    return [first, second];
}

// ✅ Both inferred
pair("hello", 42); // T = string, U = number

// ✅ Explicit T, inferred U
pair("hello", 42); // T = string, U = number

// ✅ Both explicit
pair("hello", 42);

Part 8: Real-World Examples

Example 1: API Response Handler

interface ApiResponse {
    status: number;
    data: T;
    message: string;
}

function handleApiResponse(response: ApiResponse): T {
    if (response.status !== 200) {
        throw new Error(response.message);
    }
    return response.data;
}

// Usage:
type User = { id: number; name: string };
const response: ApiResponse = {
    status: 200,
    data: { id: 1, name: "Alice" },
    message: "Success"
};

const user = handleApiResponse(response);
// user type: User (TypeScript knows!)
