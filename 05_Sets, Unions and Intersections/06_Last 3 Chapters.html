OBJECTS AND ARRAYS

COMPLETE GUIDE TO UNIONS & INTERSECTIONS WITH COMPLEX TYPES

This is where TypeScript's type system becomes truly powerful. Understanding how unions and intersections work with complex types like objects and arrays will unlock the 1% knowledge.

Part 1: Unions of Objects (OR)

1.1 Basic Object Union

// 👇 A type that can be EITHER a Dog OR a Cat
type Dog = {
    type: "dog";
    name: string;
    bark(): void;
};

type Cat = {
    type: "cat";
    name: string;
    meow(): void;
};

type Pet = Dog | Cat;

// ✅ Valid pets:
const dog: Pet = {
    type: "dog",
    name: "Rex",
    bark: () => console.log("Woof!")
};

const cat: Pet = {
    type: "cat",
    name: "Whiskers",
    meow: () => console.log("Meow!")
};

// ❌ Invalid:
const invalid: Pet = {
    type: "dog",
    name: "Rex",
    meow: () => console.log("Meow!") // ❌ Error: missing bark
};

1.2 What You CAN Access in a Union

type Dog = {
    type: "dog";
    name: string;
    bark(): void;
};

type Cat = {
    type: "cat";
    name: string;
    meow(): void;
};

type Pet = Dog | Cat;

function handlePet(pet: Pet) {
    // ✅ YOU CAN ONLY ACCESS PROPERTIES THAT EXIST IN BOTH!
    console.log(pet.type); // ✅ "dog" | "cat" (exists in both)
    console.log(pet.name); // ✅ string (exists in both)
    
    // ❌ CANNOT ACCESS PROPERTIES UNIQUE TO EITHER:
    // pet.bark(); // ❌ Error: Property 'bark' does not exist on type 'Pet'
    // pet.meow(); // ❌ Error: Property 'meow' does not exist on type 'Pet'
    
    // ✅ Must check first (Type Narrowing):
    if (pet.type === "dog") {
        pet.bark(); // ✅ Now TypeScript knows it's a Dog
    } else {
        pet.meow(); // ✅ Now TypeScript knows it's a Cat
    }
}

1.3 The Golden Rule for Unions

┌─────────────────────────────────────────────────────────────────────┐
│                    UNION GOLDEN RULE                               │
├─────────────────────────────────────────────────────────────────────┤
│                                                                     │
│  In a UNION (|), you can ONLY access properties that               │
│  are COMMON to ALL types in the union.                             │
│                                                                     │
│  Pet = Dog | Cat                                                   │
│  ┌─────────────────────────────────────────────────────────────┐   │
│  │  Common: type, name  ✅ (you can access these)             │   │
│  │  Not Common: bark, meow ❌ (you CANNOT access these)       │   │
│  └─────────────────────────────────────────────────────────────┘   │
│                                                                     │
│  To access unique properties, you MUST NARROW the type.            │
│                                                                     │
└─────────────────────────────────────────────────────────────────────┘

Part 2: Intersections of Objects (AND)

2.1 Basic Object Intersection

// 👇 A type that MUST be BOTH
type HasName = { name: string };
type HasAge = { age: number };

type Person = HasName & HasAge;
// Person = { name: string; age: number }

const person: Person = {
    name: "Alice",
    age: 25
};

// ❌ Missing age:
const invalid: Person = {
    name: "Alice"
    // Error: Property 'age' is missing
};

2.2 Intersecting More Complex Objects

type Employee = {
    id: number;
    department: string;
    salary: number;
};

type Person = {
    name: string;
    age: number;
    email: string;
};

type EmployeePerson = Employee & Person;
// EmployeePerson = {
//     id: number;
//     department: string;
//     salary: number;
//     name: string;
//     age: number;
//     email: string;
// }

const employee: EmployeePerson = {
    id: 1,
    department: "Engineering",
    salary: 75000,
    name: "Alice",
    age: 30,
    email: "alice@company.com"
};

2.3 The Golden Rule for Intersections

┌─────────────────────────────────────────────────────────────────────┐
│                    INTERSECTION GOLDEN RULE                        │
├─────────────────────────────────────────────────────────────────────┤
│                                                                     │
│  In an INTERSECTION (&), you MUST have ALL properties              │
│  from ALL types in the intersection.                               │
│                                                                     │
│  Person = HasName & HasAge                                         │
│  ┌─────────────────────────────────────────────────────────────┐   │
│  │  Required: name ✅ age ✅                                   │   │
│  │  You can access BOTH properties directly!                   │   │
│  └─────────────────────────────────────────────────────────────┘   │
│                                                                     │
│  Intersection = MORE requirements = FEWER possible objects        │
│  But MORE guarantees (you know everything exists)                │
│                                                                     │
└─────────────────────────────────────────────────────────────────────┘

Part 3: Conflicting Property Types

3.1 Conflicting Types in Intersection

// 🔥 When properties conflict, the type becomes NEVER!
type A = { value: string };
type B = { value: number };

type C = A & B;
// C = { value: never } 
// No value can be both string AND number!

// ❌ This is impossible:
const c: C = {
    value: "hello" // ❌ Error: Type 'string' is not assignable to type 'never'
};

// ❌ This is also impossible:
const c2: C = {
    value: 42 // ❌ Error: Type 'number' is not assignable to type 'never'
};

3.2 Conflicting Types in Union

// 🔥 Conflicting types in UNION work differently!
type A = { value: string };
type B = { value: number };

type C = A | B;
// C = { value: string } | { value: number }

// ✅ Both are valid:
const c1: C = { value: "hello" }; // ✅ Works
const c2: C = { value: 42 };      // ✅ Works
const c3: C = { value: true };    // ❌ Error: boolean not allowed

// The key difference:
function process(c: C) {
    console.log(c.value); // Type: string | number (union of types)
    // You can access it, but it's a union type
}

3.3 Conflicting Object Shapes

// Conflicting property shapes
type A = { 
    user: { name: string; age: number };
};

type B = { 
    user: { name: string; email: string };
};

// 🔥 Intersection becomes:
type C = A & B;
// C = {
//     user: { name: string; age: number; email: string } // Combined!
// }

// ✅ This works! The shapes combine (not conflict)
const c: C = {
    user: {
        name: "Alice",
        age: 25,
        email: "alice@email.com"
    }
};

// 🔥 But if the properties conflict directly:
type D = { 
    user: string; 
};

type E = { 
    user: number; 
};

type F = D & E;
// F = { user: never } // ❌ Conflict!

const f: F = {
    user: "hello" // ❌ Error: never
};

Part 4: Unions of Arrays

4.1 Basic Array Union

// Array that can contain EITHER strings OR numbers
type StringOrNumberArray = string[] | number[];

// ✅ Valid:
const strings: StringOrNumberArray = ["a", "b", "c"]; // ✅
const numbers: StringOrNumberArray = [1, 2, 3]; // ✅

// ❌ Invalid:
const mixed: StringOrNumberArray = ["a", 1, "b"]; 
// ❌ Error: Cannot mix string and number in same array

4.2 Complex Array Union

type NumberArray = number[];
type StringArray = string[];
type BooleanArray = boolean[];

type ArrayUnion = NumberArray | StringArray | BooleanArray;

// ✅ Valid:
let data: ArrayUnion = [1, 2, 3];        // ✅
data = ["a", "b", "c"];                  // ✅
data = [true, false, true];              // ✅
data = [1, "a", true];                   // ❌ Error: Mixed types!

Part 5: Intersections of Arrays

5.1 Basic Array Intersection

// 🔥 Intersection of arrays = Arrays that fit BOTH types
type NumberArray = number[];
type StringArray = string[];

type ArrayIntersection = NumberArray & StringArray;
// ArrayIntersection = never[] (empty set)
// No array can be BOTH number[] and string[]!

// ❌ This is impossible:
const arr: ArrayIntersection = [1, 2, 3]; // ❌ Error

5.2 Intersection of Array Types That Overlap

// 🔥 Intersection works when types overlap
type User = { name: string; age: number };
type Admin = { name: string; role: string };

type UserList = User[];
type AdminList = Admin[];

type UserAdminList = UserList & AdminList;
// UserAdminList = (User & Admin)[]
// = { name: string; age: number; role: string }[]

const users: UserAdminList = [
    { name: "Alice", age: 25, role: "admin" },
    { name: "Bob", age: 30, role: "moderator" }
];

5.3 Mixed Arrays vs Arrays of Unions

// 🔥 Array of Union (can contain different types)
type ArrayOfUnion = (string | number)[];

// 🔥 Union of Arrays (can be one type OR another)
type UnionOfArrays = string[] | number[];

// KEY DIFFERENCE:
// ArrayOfUnion:     ["a", 1, "b", 2] ✅ (mixed types in one array)
// UnionOfArrays:    ["a", 1, "b", 2] ❌ (mixed types not allowed)

// Visual:
// ArrayOfUnion = { ["a", 1, "b"], ["a", "b"], [1, 2] }  // BIGGER
// UnionOfArrays = { ["a", "b"], [1, 2] }                 // SMALLER

// Example:
let mixed: (string | number)[] = ["a", 1, "b", 2]; // ✅ Works
let unionArr: string[] | number[] = ["a", 1, "b", 2]; // ❌ Error
