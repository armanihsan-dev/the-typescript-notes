TYPES AS SETS OF JAVASCRIPT VALUES

THE MOST IMPORTANT CONCEPT IN TYPESCRIPT. ONCE YOU TRULY UNDERSTAND THIS, EVERYTHING ELSE CLICKS INTO PLACE.

Part 1: The Core Truth

Every TypeScript type is just a SET of JavaScript values that are allowed to be used where that type is expected.

┌─────────────────────────────────────────────────────────────────────┐
│                    TYPES = SETS OF VALUES                           │
├─────────────────────────────────────────────────────────────────────┤
│                                                                     │
│  In TypeScript:                                                     │
│  ┌─────────────────────────────────────────────────────────────┐   │
│  │  Every type represents a SET of possible values            │   │
│  │                                                             │   │
│  │  string = { "hello", "world", "a", "b", ... }             │   │
│  │  number = { 1, 2, 3, 3.14, -5, ... }                     │   │
│  │  boolean = { true, false }                                 │   │
│  │  null = { null }                                           │   │
│  │  undefined = { undefined }                                 │   │
│  │  never = { } (empty set)                                  │   │
│  │  unknown = { ALL POSSIBLE VALUES }                        │   │
│  │  any = { ALL POSSIBLE VALUES (with type checking off) }  │   │
│  └─────────────────────────────────────────────────────────────┘   │
│                                                                     │
└─────────────────────────────────────────────────────────────────────┘

Part 2: Built-in Types as Sets

1. string - The Set of All Strings

// string = { "hello", "world", "a", "b", "", ... }
// It includes EVERY possible string value

let name: string = "Alice";        // ✅ In the set
let city: string = "New York";     // ✅ In the set
let empty: string = "";            // ✅ In the set
let number: string = "123";        // ✅ In the set (it's a string)
let notAString: string = 42;       // ❌ NOT in the set

// TypeScript's error:
// Type 'number' is not assignable to type 'string'
// "42" is NOT in the set of strings

2. number - The Set of All Numbers

// number = { 1, 2, 3, 3.14, -5, Infinity, NaN, ... }
// It includes EVERY possible number value

let age: number = 25;              // ✅ In the set
let price: number = 99.99;         // ✅ In the set
let negative: number = -10;        // ✅ In the set
let infinity: number = Infinity;   // ✅ In the set
let nan: number = NaN;             // ✅ In the set
let notANumber: number = "25";     // ❌ NOT in the set

// TypeScript's error:
// Type 'string' is not assignable to type 'number'

3. boolean - The Set of Exactly Two Values

// boolean = { true, false }
// That's it! Only two possible values.

let isActive: boolean = true;      // ✅ In the set
let isDone: boolean = false;       // ✅ In the set
let maybe: boolean = "true";       // ❌ NOT in the set
let maybe2: boolean = 1;           // ❌ NOT in the set

// TypeScript's error:
// Type 'string' is not assignable to type 'boolean'

4. null and undefined - Singleton Sets

// null = { null }
// undefined = { undefined }

let nothing: null = null;          // ✅ In the set
let something: null = "hello";     // ❌ NOT in the set
let notAssigned: undefined = undefined; // ✅ In the set
let assigned: undefined = "hello"; // ❌ NOT in the set

// Each has exactly ONE possible value!

5. never - The Empty Set

// never = { }
// No value exists in this set

function throwError(): never {
    throw new Error("Always throws");
}

function infiniteLoop(): never {
    while(true) {
        // Never returns
    }
}

// You can never assign a value to 'never'
let impossible: never = 5;         // ❌ NOT in the set (nothing is!)

6. unknown - The Universal Set

// unknown = { EVERY POSSIBLE VALUE }
// All values belong to this set

let anything: unknown = 5;         // ✅ In the set
anything = "hello";                // ✅ In the set
anything = true;                   // ✅ In the set
anything = null;                   // ✅ In the set
anything = undefined;              // ✅ In the set
anything = { name: "Alice" };      // ✅ In the set
anything = [1, 2, 3];              // ✅ In the set

// But you can't DO anything with it
// unknown = "Can't use this safely!"

7. any - The "I Don't Care" Set

// any = { EVERY POSSIBLE VALUE } (with type checking disabled)
// Same as unknown, but unsafe

let anything: any = 5;             // ✅ In the set
anything = "hello";                // ✅ In the set
anything.toUpperCase();            // ✅ Allowed (could crash at runtime)
anything.push(5);                  // ✅ Allowed (could crash at runtime)
// Type checking is OFF for 'any'

Part 3: Literal Types as Sets

1. String Literal Type - Set with One Value

// "hello" = { "hello" }
// Only ONE value can be in this set

let greeting: "hello" = "hello";   // ✅ In the set
let wrong: "hello" = "goodbye";    // ❌ NOT in the set

// TypeScript's error:
// Type '"goodbye"' is not assignable to type '"hello"'

2. Number Literal Type - Set with One Value

// 42 = { 42 }
// Only ONE value can be in this set

let answer: 42 = 42;               // ✅ In the set
let wrong: 42 = 43;                // ❌ NOT in the set

// TypeScript's error:
// Type '43' is not assignable to type '42'

3. Boolean Literal Type - Set with One Value

// true = { true }
// false = { false }

let isTrue: true = true;           // ✅ In the set
let isFalse: true = false;         // ❌ NOT in the set

Part 4: Union Types as Set Union

In Set Theory

A ∪ B = { x | x ∈ A OR x ∈ B }

In TypeScript

// "hello" ∪ 42 = { "hello", 42 }
type Greeting = "hello" | 42;

let valid1: Greeting = "hello";    // ✅ In the set
let valid2: Greeting = 42;         // ✅ In the set
let invalid: Greeting = "goodbye"; // ❌ NOT in the set
let invalid2: Greeting = 43;       // ❌ NOT in the set

// string ∪ number = ALL strings AND ALL numbers
type StringOrNumber = string | number;

let s: StringOrNumber = "hello";   // ✅ In the set
let n: StringOrNumber = 42;        // ✅ In the set
let b: StringOrNumber = true;      // ❌ NOT in the set (boolean not included)

// Color set = { "red", "green", "blue" }
type Color = "red" | "green" | "blue";

let c1: Color = "red";             // ✅ In the set
let c2: Color = "green";           // ✅ In the set
let c3: Color = "blue";            // ✅ In the set
let c4: Color = "yellow";          // ❌ NOT in the set

Visual: Union as Set Union

├─────────────────────────────────────────────────────────────────────┤

│                                                                     │
│  Set C: { "red", "green", "blue" }                                │

Part 5: Intersection Types as Set Intersection

In Set Theory

A ∩ B = { x | x ∈ A AND x ∈ B }

In TypeScript

// Intersection of literals: only common values
type A = "hello" | "world" | 42;
type B = "world" | 42 | true;

type Intersection = A & B;
// Intersection = { "world", 42 }
// Only values that are in BOTH sets

let i1: Intersection = "world";    // ✅ In the set
let i2: Intersection = 42;         // ✅ In the set
let i3: Intersection = "hello";    // ❌ NOT in the set (only in A)
let i4: Intersection = true;       // ❌ NOT in the set (only in B)

// Intersection of object types: combine properties
type HasName = { name: string };
type HasAge = { age: number };

type Person = HasName & HasAge;
// Person = { name: string; age: number }
// Must have BOTH properties

let person1: Person = {
    name: "Alice",                 // ✅ From HasName
    age: 25                        // ✅ From HasAge
};

let person2: Person = {
    name: "Bob"                    // ❌ Missing age
};

Visual: Intersection as Set Intersection

│                    INTERSECTION AS SET INTERSECTION                 │
├─────────────────────────────────────────────────────────────────────┤
│                                                                     │
│  Set A: { "hello", "world", 42 }                                   │
│  ┌─────────────────────────────────┐                               │
│  │  "hello"   │  "world" │  42    │                               │
│  └─────────────────────────────────┘                               │
│                                                                     │
│  Set B: { "world", 42, true }                                      │
│  ┌─────────────────────────────────┐                               │
│  │  "world"   │  42     │  true   │                               │
│  └─────────────────────────────────┘                               │
│                                                                     │
│  A ∩ B = { "world", 42 }                                          │
│  ┌────────────────────┐                                            │
│  │  "world"  │  42    │                                            │
│  └────────────────────┘                                            │
│                                                                     │
└─────────────────────────────────────────────────────────────────────┘

Part 6: Subtypes as Subsets

In Set Theory

A ⊆ B if every element of A is also in B

In TypeScript

// "hello" is a subset of string
// Every value that is "hello" is also a string

let specific: "hello" = "hello";   // ✅ "hello" is in the set
let general: string = specific;    // ✅ Allowed! "hello" is in string

// But you can't go the other way
let general2: string = "world";    // ✅ "world" is in the set
let specific2: "hello" = general2; // ❌ "world" is NOT in { "hello" }

// Visual:
// string = { "hello", "world", "a", "b", ... }
// "hello" = { "hello" }
// "hello" ⊆ string (subset)

// Object types: Subtypes have MORE properties
type Animal = { name: string };
type Dog = Animal & { breed: string };

// Dog is a subset of Animal (every dog is an animal)
// Animal is NOT a subset of Dog (not every animal is a dog)

let dog: Dog = {
    name: "Rex",
    breed: "Golden Retriever"
};

let animal: Animal = dog;          // ✅ Allowed (Dog ⊆ Animal)
let dog2: Dog = animal;            // ❌ NOT allowed (Animal ⊄ Dog)
// Error: Property 'breed' is missing

Part 8: Why This Matters

1. Type Checking = Set Membership

// TypeScript is asking: "Is this value in this set?"

function greet(name: string): string {
    return `Hello ${name}`;
}

greet("Alice");  // ✅ "Alice" ∈ string
greet(42);       // ❌ 42 ∉ string

// TypeScript's internal logic:
// if (value ∈ type) { OK } else { ERROR }

2. Assignability = Set Containment

// Can we assign value to variable?

let specific: "hello" = "hello";
let general: string = specific;
// ✅ "hello" ⊆ string (specific is subset of string)
// "Is every value in 'specific' also in 'string'?" → YES

let general2: string = "world";
let specific2: "hello" = general2;
// ❌ "world" ∉ "hello"
// "Is every value in 'string' also in 'hello'?" → NO

3. Union = Set Union (More Values)

// Union increases the set size
type A = "hello" | "world";
// A = { "hello", "world" }

type B = A | 42;
// B = { "hello", "world", 42 }

// More values allowed = fewer guarantees
function process(value: A | number) {
    // We can't guarantee it's a string OR a number
    // Must check first
}

4. Intersection = Set Intersection (Fewer Values)

// Intersection decreases the set size
type A = { name: string };
type B = { age: number };
type C = A & B;
// C = { name: string; age: number }

// Fewer values allowed = more guarantees
function process(person: A & B) {
    // We can guarantee BOTH name AND age exist
    console.log(person.name);  // ✅ Guaranteed
    console.log(person.age);   // ✅ Guaranteed
}

One-Line Summary

Every TypeScript type is a set of JavaScript values, and type checking is just checking if a value belongs to that set.
