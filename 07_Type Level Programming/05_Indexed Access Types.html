WHAT ARE INDEXED ACCESS TYPES?

Definition:

Indexed Access Types let you look up the type of a specific property inside another type. It's like using [] to access a property on an object — but at the type level.

The One-Line Summary

type ValueType = SomeType["propertyName"];
// Gets the type of "propertyName" inside SomeType

🧠 The Mental Model

┌─────────────────────────────────────────────────────────────────┐
│                    INDEXED ACCESS TYPES                          │
├─────────────────────────────────────────────────────────────────┤
│                                                                  │
│  VALUE LEVEL:              TYPE LEVEL:                          │
│  ─────────────            ────────────                          │
│                                                                  │
│  const user = {            type User = {                        │
│      name: "Alice",            name: string;                    │
│      age: 25                   age: number;                     │
│  };                        };                                   │
│                                                                  │
│  user["name"]              User["name"]                         │
│  → "Alice"                 → string                             │
│                                                                  │
│  You access a VALUE.       You access a TYPE.                   │
│                                                                  │
└─────────────────────────────────────────────────────────────────┘

Part 1: Basic Syntax

type ValueType = SomeType["key"];

Simple Example

type User = {
    name: string;
    age: number;
    email: string;
};

type NameType = User["name"];
// NameType = string

type AgeType = User["age"];
// AgeType = number

type EmailType = User["email"];
// EmailType = string

Part 2: How It Works

Step by Step

// Step 1: Define an object type
type User = {
    name: string;
    age: number;
    isActive: boolean;
};

// Step 2: Access a specific property type
type NameType = User["name"];     // string
type AgeType = User["age"];       // number
type ActiveType = User["isActive"]; // boolean

// Step 3: Use it anywhere
const name: NameType = "Alice";   // ✅ Works
const age: AgeType = 25;          // ✅ Works
const active: ActiveType = true;  // ✅ Works

Visual

┌─────────────────────────────────────────────────────────────────┐
│                    HOW IT WORKS                                 │
├─────────────────────────────────────────────────────────────────┤
│                                                                  │
│  type User = {                                                 │
│      name: string;      ← User["name"] = string                │
│      age: number;       ← User["age"] = number                 │
│      isActive: boolean; ← User["isActive"] = boolean           │
│  };                                                            │
│                                                                  │
│  You access the TYPE of each property using ["property"].     │
│                                                                  │

Part 3: Indexed Access with Unions

type User = {
    name: string;
    age: number;
    email: string;
};

// Union of keys
type StringOrNumber = User["name" | "age"];
// StringOrNumber = string | number

type AllTypes = User["name" | "age" | "email"];
// AllTypes = string | number | string = string | number

Part 4: Indexed Access with keyof

T[keyof T] = All Value Types

type User = {
    name: string;
    age: number;
    isActive: boolean;
};

type AllValues = User[keyof User];
// AllValues = string | number | boolean

How It Works

Part 5: Indexed Access with Arrays

Array Element Type

type Colors = string[];

type ColorType = Colors[number];
// ColorType = string

Tuple Element Types

type Point = [number, number];

type X = Point[0]; // number
type Y = Point[1]; // number

type Mixed = [string, number, boolean];

type First = Mixed[0];  // string
type Second = Mixed[1]; // number
type Third = Mixed[2];  // boolean

Array of Objects

type Users = { name: string; age: number }[];

type UserType = Users[number];
// UserType = { name: string; age: number }

Part 6: Indexed Access with Nested Objects

type User = {
    name: string;
    address: {
        street: string;
        city: string;
        zip: string;
    };
};

// Access nested types
type AddressType = User["address"];
// AddressType = { street: string; city: string; zip: string }

type StreetType = User["address"]["street"];
// StreetType = string

type CityType = User["address"]["city"];
// CityType = string

Part 7: Indexed Access with Generic Constraints

The Classic Pattern

function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
    return obj[key];
}

const user = { name: "Alice", age: 25 };

const name = getProperty(user, "name"); // string
const age = getProperty(user, "age");   // number
// getProperty(user, "email"); // ❌ Error

Why T[K] Works

┌─────────────────────────────────────────────────────────────────┐
│                    T[K] EXPLAINED                               │
├─────────────────────────────────────────────────────────────────┤
│                                                                  │
│  T = { name: string; age: number }                             │
│  K = "name"                                                    │
│                                                                  │
│  T[K] = T["name"] = string                                     │
│                                                                  │
│  The return type DEPENDS on the key you pass.                  │
│                                                                  │
└─────────────────────────────────────────────────────────────────┘

Part 8: Indexed Access with typeof

const user = {
    name: "Alice",
    age: 25,
    email: "alice@email.com"
};

// Get type of a property from a value
type NameType = (typeof user)["name"];
// NameType = string

type AgeType = (typeof user)["age"];
// AgeType = number

type UserType = typeof user;
// UserType = { name: string; age: number; email: string }

type AllValues = (typeof user)[keyof typeof user];
// AllValues = string | number

Part 9: Practical Use Cases

Use Case 1: Type-Safe Getters

function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
    return obj[key];
}

const user = { name: "Alice", age: 25 };
const name = getProperty(user, "name"); // string
const age = getProperty(user, "age");   // number

Use Case 2: Extract Specific Types

type ApiResponse = {
    status: number;
    data: {
        user: {
            id: number;
            name: string;
        };
        posts: {
            id: number;
            title: string;
        }[];
    };
};

// Extract the user type
type User = ApiResponse["data"]["user"];
// User = { id: number; name: string }

// Extract the posts type
type Posts = ApiResponse["data"]["posts"];
// Posts = { id: number; title: string }[]

Use Case 3: Array Element Type Extraction

type Users = {
    id: number;
    name: string;
}[];

type UserType = Users[number];

Users is an array type. Each element must be an object with a numeric id and a string name.
type UserType = Users[number];

The [number] is TypeScript’s indexed access type syntax. Here, number represents any numeric array index, so Users[number] means “the type of an element at a numeric index in this array.”

Therefore, UserType is the element type of Users:

// UserType = { id: number; name: string }

// Real-world use:
function processUsers(users: Users): UserType[] {
    return users.map(user => ({ id: user.id, name: user.name }));
}

Use Case 4: Type-Safe Object Updates

function updateProperty<T, K extends keyof T>(
    obj: T,
    key: K,
    value: T[K]
): T {
    return { ...obj, [key]: value };
}

const user = { name: "Alice", age: 25 };

const updated = updateProperty(user, "name", "Bob"); // ✅
// updated = { name: "Bob", age: 25 }

// ❌ Error: value must be number for "age"
updateProperty(user, "age", "old");
