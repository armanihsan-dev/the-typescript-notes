INDEX SIGNATURES IN TYPESCRIPT — COMPLETE GUIDE

🎯 What Are Index Signatures?

Definition:

An index signature is a way to define a type for an object that can have any number of properties with dynamic keys, but all values must follow the same type.

The One-Line Summary

type Dictionary = {
    [key: string]: string;
};
// Any string key, value must be string

🧠 The Mental Model

┌─────────────────────────────────────────────────────────────────┐
│                    INDEX SIGNATURE                              │
├─────────────────────────────────────────────────────────────────┤
│                                                                  │
│  REGULAR OBJECT TYPE:              INDEX SIGNATURE:             │
│  ───────────────────               ─────────────────            │
│                                                                  │
│  type User = {                     type Dictionary = {          │
│      name: string;                     [key: string]: string;   │
│      age: number;                  };                           │
│  };                                                             │
│                                                                  │
│  FIXED keys (name, age).           DYNAMIC keys (any string).  │
│  You know them in advance.         You don't know them.        │
│                                                                  │
└─────────────────────────────────────────────────────────────────┘

Part 1: Basic Syntax

String Index Signature

type Dictionary = {
    [key: string]: string;
};

const translations: Dictionary = {
    hello: "hola",
    goodbye: "adiós",
    thankYou: "gracias"
};

Number Index Signature

type NumberDictionary = {
    [index: number]: string;
};

const fruits: NumberDictionary = {
    0: "apple",
    1: "banana",
    2: "orange"
};

Part 2: How It Works

The Syntax Breakdown

type Dictionary = {
    [key: string]: string;
//   ↑      ↑        ↑
//   |      |        |
//   |      |        Value type
//   |      Key type
//   Parameter name (can be anything)
};

What It Means

┌─────────────────────────────────────────────────────────────────┐
│                    HOW IT WORKS                                 │
├─────────────────────────────────────────────────────────────────┤
│                                                                  │
│  [key: string]: string                                          │
│                                                                  │
│  Means:                                                         │
│  • ANY string key is allowed                                    │
│  • The value for ANY key must be a string                      │
│  • You can add as many properties as you want                  │
│  • You can't add a property with a different value type        │
│                                                                  │
│  ┌──────────────────────────────────────────────────────────┐   │
│  │  ✅ Allowed:                                             │   │
│  │  { hello: "hola", goodbye: "adiós" }                    │   │
│  │                                                         │   │
│  │  ❌ Not allowed:                                         │   │
│  │  { hello: "hola", count: 42 }                           │   │
│  │  // count is number, but all values must be string     │   │
│  └──────────────────────────────────────────────────────────┘   │
│                                                                  │
└─────────────────────────────────────────────────────────────────┘

Part 3: Key Type Rules

String vs Number vs Symbol

// ✅ String keys
type StringKeys = {
    [key: string]: number;
};

// ✅ Number keys
type NumberKeys = {
    [index: number]: string;
};

// ✅ Symbol keys
type SymbolKeys = {
    [key: symbol]: boolean;
};

JavaScript Behavior (Important!)

// 🔥 In JavaScript, number keys are converted to strings!
const obj = {
    0: "apple",
    1: "banana"
};
console.log(Object.keys(obj)); // ["0", "1"] (strings!)

// 🔥 TypeScript allows both string and number index signatures
type Mixed = {
    [key: string]: string;
    [index: number]: string;
};

Part 4: Index Signature + Regular Properties

The Rule

type User = {
    name: string;              // Fixed property
    age: number;               // Fixed property
    [key: string]: string | number;  // Index signature
};

// ✅ Works: All properties match the index type
const user: User = {
    name: "Alice",
    age: 25,
    email: "alice@email.com"  // ✅ string matches
};

// ❌ Fails: A property doesn't match
type User2 = {
    name: string;
    age: number;
    [key: string]: string;    // Only strings allowed
};
// ❌ Error: 'age' of type 'number' is not assignable to 'string'

Visual

┌─────────────────────────────────────────────────────────────────┐
│                    INDEX + REGULAR PROPERTIES                   │
├─────────────────────────────────────────────────────────────────┤
│                                                                  │
│  Rule: ALL regular properties MUST match the index signature.  │
│                                                                  │
│  ┌──────────────────────────────────────────────────────────┐   │
│  │  type User = {                                          │   │
│  │      name: string;        // string ✅                  │   │
│  │      age: number;         // number ✅                  │   │
│  │      [key: string]: string | number;  // allows both    │   │
│  │  }                                                      │   │
│  └──────────────────────────────────────────────────────────┘   │
│                                                                  │
│  ┌──────────────────────────────────────────────────────────┐   │
│  │  type User2 = {                                         │   │
│  │      name: string;        // string ✅                  │   │
│  │      age: number;         // number ❌ (not string!)    │   │
│  │      [key: string]: string;  // only string allowed     │   │
│  │  }                                                      │   │

Part 5: Readonly Index Signatures

type ReadonlyDictionary = {
    readonly [key: string]: string;
};

const dict: ReadonlyDictionary = {
    hello: "hola"
};

dict.hello = "bonjour"; // ❌ Error: readonly
dict.newKey = "value";  // ❌ Error: readonly

Part 6: Index Signatures with Record

Record<K, V> Uses Index Signatures

// These are equivalent:
type Dictionary1 = {
    [key: string]: string;
};

type Dictionary2 = Record<string, string>;

// Both allow:
const d1: Dictionary1 = { hello: "hola" };
const d2: Dictionary2 = { hello: "hola" };

Record with Union Keys

type Status = "pending" | "approved" | "rejected";

type StatusMessages = Record<Status, string>;
// Equivalent to:
// {
//     pending: string;
//     approved: string;
//     rejected: string;
// }

const messages: StatusMessages = {
    pending: "Waiting...",
    approved: "Done!",
    rejected: "Failed"
};

Part 7: Practical Use Cases

Use Case 1: Translation Dictionary

type Translation = {
    [key: string]: string;
};

const englishToSpanish: Translation = {
    hello: "hola",
    goodbye: "adiós",
    thankYou: "gracias"
};

// Type-safe access
const word: string = englishToSpanish["hello"];

Use Case 2: Cache Storage

type Cache = {
    [key: string]: {
        value: unknown;
        expiresAt: number;
    };
};

const cache: Cache = {
    "user:1": { value: { name: "Alice" }, expiresAt: Date.now() + 3600000 },
    "user:2": { value: { name: "Bob" }, expiresAt: Date.now() + 3600000 }
};

Use Case 4: Dynamic Form Data

type FormData = {
    [fieldName: string]: string | number | boolean;
};

const formData: FormData = {
    username: "alice",
    age: 25,
    isActive: true
};

Part 9: Common Mistakes

Mistake 1: Index Value Type Too Narrow

// ❌ Error: age doesn't match string
type User = {
    name: string;
    age: number;
    [key: string]: string;
};

// ✅ Fix: Widen the index type
type User2 = {
    name: string;
    age: number;
    [key: string]: string | number;
};

Mistake 2: Using Index Signature When You Need Specific Keys

// ❌ Bad: Loses type safety
type Config = {
    [key: string]: string;
};

// ✅ Better: Use specific keys
type Config2 = {
    apiUrl: string;
    timeout: string;
};

Mistake 3: Forgetting undefined in Lookups

type Dict = { [key: string]: string };

const dict: Dict = { hello: "hola" };

const value = dict["nonexistent"]; // string
// But at runtime, it's undefined!

// ✅ Fix: Add undefined to index type
type SafeDict = { [key: string]: string | undefined };

const safeDict: SafeDict = { hello: "hola" };
const safeValue = safeDict["nonexistent"]; // string | undefined
