Branded Types in TypeScript

Complete Guide

🎯 What Are Branded Types?

Definition:

A branded type (also called opaque type or nominal type) is a type that looks like a primitive type (like string or number) but is treated as a distinct type by TypeScript. It prevents you from accidentally mixing up values that have the same underlying type but different meanings.

The One-Line Summary

type UserId = string & { __brand: "UserId" };
type ProductId = string & { __brand: "ProductId" };
// Both are strings at runtime, but TypeScript treats them as DIFFERENT types

🧠 The Problem It Solves

The Problem: Structural Typing

TypeScript uses structural typing — if two types have the same shape, they're compatible. 

// ❌ Both are just strings
type UserId = string;
type ProductId = string;

function getUser(id: UserId): User { /* ... */ }
function getProduct(id: ProductId): Product { /* ... */ }

const userId: UserId = "user-123";
const productId: ProductId = "prod-456";

// ❌ These should NOT work, but they do!
getUser(productId);   // Bug! Passing a product ID to getUser
getProduct(userId);   // Bug! Passing a user ID to getProduct

Part 2: The Solution — Branded Types

Basic Syntax

type Brand<T, B> = T & { __brand: B };

type UserId = Brand<string, "UserId">;
type ProductId = Brand<string, "ProductId">;

How It Works

┌─────────────────────────────────────────────────────────────────┐
│                    HOW BRANDING WORKS                           │
├─────────────────────────────────────────────────────────────────┤
│                                                                  │
│  UserId = string & { __brand: "UserId" }                        │
│                                                                  │
│  At RUNTIME:                                                    │
│  ┌──────────────────────────────────────────────────────────┐   │
│  │  It's just a string!                                     │   │
│  │  The __brand property doesn't exist                      │   │
│  │  No runtime overhead                                     │   │
│  └──────────────────────────────────────────────────────────┘   │
│                                                                  │
│  At COMPILE TIME:                                               │
│  ┌──────────────────────────────────────────────────────────┐   │
│  │  TypeScript sees:                                        │   │
│  │  - UserId has a __brand property of "UserId"            │   │
│  │  - ProductId has a __brand property of "ProductId"     │   │
│  │  - They are NOT compatible!                             │   │

Part 3: Creating Branded Types

Method 1: Using & (Intersection)

type UserId = string & { readonly __brand: "UserId" };
type ProductId = string & { readonly __brand: "ProductId" };

Method 2: Using a Generic Helper

type Brand<T, B> = T & { readonly __brand: B };

type UserId = Brand<string, "UserId">;
type ProductId = Brand<string, "ProductId">;
type Email = Brand<string, "Email">;

Method 3: Using unique symbol (Strongest)

declare const brand: unique symbol;

type Brand<T, B> = T & { readonly [brand]: B };

type UserId = Brand<string, "UserId">;
type ProductId = Brand<string, "ProductId">;

Part 4: Creating Branded Values

The Problem

You can't just assign a regular string to a branded type:

type UserId = string & { readonly __brand: "UserId" };

// ❌ Error: Type 'string' is not assignable to type 'UserId'
const id: UserId = "user-123";

Solution 1 : using assertion:

UserId is more than just a string in TypeScript:

type UserId = string & { __brand: "UserId" };

It means a value must be both a string and have the brand type { __brand: "UserId" }. The plain value 'user-123' is inferred as a string; it does not have that brand, so TypeScript rejects the assignment. 

let user1: UserId = 'user-123'; // Error: plain string has no UserId brand

Adding as UserId is a type assertion. It tells TypeScript, “Trust me; treat this value as a UserId” 

let user1: UserId = 'user-123' as UserId;

Solution 2: Factory Functions

type UserId = string & { readonly __brand: "UserId" };

// ✅ Factory function to create a UserId
function createUserId(id: string): UserId {
    return id as UserId;
}

const userId = createUserId("user-123");
// userId is UserId

With Validation

type Email = string & { readonly __brand: "Email" };

function createEmail(email: string): Email {
    if (!email.includes("@")) {
        throw new Error("Invalid email");
    }
    return email as Email;
}

const email = createEmail("alice@email.com"); // ✅
// const invalid = createEmail("not-an-email"); // ❌ Throws error

Part 5: How Branded Types Prevent Bugs

Example: UserId vs ProductId

type UserId = string & { readonly __brand: "UserId" };
type ProductId = string & { readonly __brand: "ProductId" };

function createUserId(id: string): UserId {
    return id as UserId;
}

function createProductId(id: string): ProductId {
    return id as ProductId;
}

function getUser(id: UserId): User {
    // ...
    return {} as User;
}

function getProduct(id: ProductId): Product {
    // ...
    return {} as Product;
}

const userId = createUserId("user-123");
const productId = createProductId("prod-456");

// ✅ These work
getUser(userId);
getProduct(productId);

// ❌ These FAIL at compile time!
getUser(productId);    // Error: ProductId is not assignable to UserId
getProduct(userId);    // Error: UserId is not assignable to ProductId

Part 6: Real-World Use Cases

Use Case 1: Validated Strings

type Email = string & { readonly __brand: "Email" };
type Phone = string & { readonly __brand: "Phone" };
type URL = string & { readonly __brand: "URL" };

function createEmail(email: string): Email | null {
    return email.includes("@") ? (email as Email) : null;
}

function createPhone(phone: string): Phone | null {
    return /^\d{3}-\d{3}-\d{4}$/.test(phone) ? (phone as Phone) : null;
}

function sendEmail(to: Email, body: string): void {
    // ...
}

const email = createEmail("alice@email.com");
const phone = createPhone("123-456-7890");

if (email) {
    sendEmail(email, "Hello!"); // ✅
}
// sendEmail(phone, "Hello!"); // ❌ Error

Use Case 2: Currency Safety

type USD = number & { readonly __brand: "USD" };
type EUR = number & { readonly __brand: "EUR" };
type GBP = number & { readonly __brand: "GBP" };

function usd(amount: number): USD {
    return amount as USD;
}

function eur(amount: number): EUR {
    return amount as EUR;
}

function addUSD(a: USD, b: USD): USD {
    return (a + b) as USD;
}

const price1 = usd(100);
const price2 = usd(50);
const price3 = eur(75);

addUSD(price1, price2); // ✅
// addUSD(price1, price3); // ❌ Error: Can't add USD to EUR

Use Case 3: Database IDs

type UserId = number & { readonly __brand: "UserId" };
type PostId = number & { readonly __brand: "PostId" };
type CommentId = number & { readonly __brand: "CommentId" };

function fetchUser(id: UserId): Promise<User> {
    // ...
    return Promise.resolve({} as User);
}

function fetchPost(id: PostId): Promise<Post> {
    // ...
    return Promise.resolve({} as Post);
}

const userId = 1 as UserId;
const postId = 2 as PostId;

fetchUser(userId);   // ✅
fetchPost(postId);   // ✅
// fetchUser(postId); // ❌ Error

Part 7: Branded Types with Generics

Generic Brand Helper

type Brand<T, B extends string> = T & { readonly __brand: B };

type UserId = Brand<string, "UserId">;
type ProductId = Brand<string, "ProductId">;
type OrderId = Brand<number, "OrderId">;
type Price = Brand<number, "Price">;
type Discount = Brand<number, "Discount">;

Factory Pattern

type Brand<T, B extends string> = T & { readonly __brand: B };

function createBrand<T, B extends string>(value: T): Brand<T, B> {
    return value as Brand<T, B>;
}

const userId = createBrand<string, "UserId">("user-123");
const productId = createBrand<string, "ProductId">("prod-456");

// ✅ Type-safe
function getUser(id: Brand<string, "UserId">): void {}
getUser(userId); // ✅
// getUser(productId); // ❌ Error

Part 8: Unbranding

Extracting the Base Type

type UserId = string & { readonly __brand: "UserId" };

// Extract the underlying type
type Unbrand<T> = T extends Brand<infer U, any> ? U : T;

type BaseType = Unbrand<UserId>; // string

function getRawValue(id: UserId): string {
    return id as string; // Unbrand by casting
}

Validation Before Unbranding

function isValidUserId(id: string): id is UserId {
    return id.startsWith("user-");
}

const rawId = "user-123";
if (isValidUserId(rawId)) {
    // rawId is now UserId
    getUser(rawId); // ✅
}

Part 9: Common Mistakes

Mistake 1: Using as Without Validation

// ❌ Dangerous: No validation
const userId = "anything" as UserId;

// ✅ Better: Validate first
function createUserId(id: string): UserId {
    if (!id.startsWith("user-")) {
        throw new Error("Invalid user ID");
    }
    return id as UserId;
}

Mistake 2: Forgetting to Export Factory Functions

// types.ts
export type UserId = string & { readonly __brand: "UserId" };
export function createUserId(id: string): UserId {
    return id as UserId;
}

// ✅ Always export the factory too

Mistake 3: Using interface Instead of type

// ❌ Doesn't work as expected
interface UserId extends String {}

// ✅ Use type alias
type UserId = string & { readonly __brand: "UserId" };
